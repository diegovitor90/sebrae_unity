# Etapa 4: arco e flecha. Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** puxar a corda do arco com a mão livre (segundo `XRGrabInteractable`), disparar uma flecha com gravidade real e pontuar no alvo pelo anel atingido.

**Architecture:** as regras ficam em duas classes C# puras (`ArcheryScore` para anéis e placar, `ArcheryDraw` para puxada e velocidade), testadas em EditMode. O componente `ArcheryRange` recebe do `InternshipDay` as peças já montadas e cuida do alvo, do placar, do disparo, do modo PC e da recarga. O componente `ArcheryString` cuida do pegador e da corda visual. `ArcheryArrow` voa com gravidade própria e um raio por passo. A textura do alvo é regenerada a partir da mesma tabela de raios da pontuação. Uma verificação em Play (`ArcheryCheck`) é escrita antes de cada componente.

**Tech Stack:** Unity 6000.0.83f1, Built-in Render Pipeline, XR Interaction Toolkit 3.0.11, PhysX, C#.

```csharp
using System;

namespace SebraeLab.Missions
{
    /// <summary>
    /// Regras do alvo do arco: anel pela distância ao centro e placar do tiro. Sem dependência de cena.
    /// Os raios são os mesmos usados para pintar a textura do alvo (AlvoTexture).
    /// </summary>
    public class ArcheryScore
    {
        /// <summary>Raio externo de cada anel, em metros, do centro para fora. A borda pertence ao anel de dentro.</summary>
        public static readonly float[] Raios = { .14f, .28f, .42f, .56f, .70f };
        public static readonly int[] PontosDosAneis = { 10, 8, 6, 4, 2 };
        static readonly string[] Rotulos = { "centro", "anel 2", "anel 3", "anel 4", "anel 5" };

        public int Total { get; private set; }
        public int Tiros { get; private set; }
        public int UltimoPontos { get; private set; }
        public string UltimoRotulo { get; private set; } = "nenhum";

        /// <summary>Pontos do anel atingido a esta distância do centro, ou 0 fora do último anel.</summary>
        public static int PontosDoAnel(float distancia)
        {
            int i = IndiceDoAnel(distancia);
            return i < 0 ? 0 : PontosDosAneis[i];
        }

        public void Registrar(float distancia)
        {
            if (distancia < 0f) throw new ArgumentOutOfRangeException(nameof(distancia), "A distância não pode ser negativa.");
            int i = IndiceDoAnel(distancia);
            if (i < 0) { Errou(); return; }
            Tiros++;
            UltimoPontos = PontosDosAneis[i];
            UltimoRotulo = Rotulos[i];
            Total += UltimoPontos;
        }

        public void Errou()
        {
            Tiros++;
            UltimoPontos = 0;
            UltimoRotulo = "Errou";
        }

        public string Texto
        {
            get
            {
                string ultimo = Tiros == 0 ? "nenhum" : UltimoPontos == 0 ? "Errou" : $"{UltimoPontos} ({UltimoRotulo})";
                return $"ARCO ARCADE • {Total} pontos\nÚltimo tiro: {ultimo}";
            }
        }

        static int IndiceDoAnel(float distancia)
        {
            for (int i = 0; i < Raios.Length; i++) if (distancia <= Raios[i]) return i;
            return -1;
        }
    }
}
```

```csharp
using UnityEngine;

namespace SebraeLab.Missions
{
    /// <summary>Conversão da puxada da corda (metros) em velocidade da flecha (m/s). Sem dependência de cena.</summary>
    public static class ArcheryDraw
    {
        public const float PuxadaMaxima = .5f;
        public const float PuxadaMinima = .1f;
        public const float VelocidadeMinima = 15f;
        public const float VelocidadeMaxima = 40f;

        public static float Limitar(float puxada) => Mathf.Clamp(puxada, 0f, PuxadaMaxima);

        public static bool Dispara(float puxada) => puxada >= PuxadaMinima;

        /// <summary>0 abaixo da puxada mínima; depois, interpolação linear de 15 a 40 m/s até a puxada máxima.</summary>
        public static float Velocidade(float puxada)
        {
            if (!Dispara(puxada)) return 0f;
            return Mathf.Lerp(VelocidadeMinima, VelocidadeMaxima, Mathf.InverseLerp(PuxadaMinima, PuxadaMaxima, Limitar(puxada)));
        }
    }
}
```

```csharp
using System;
using System.IO;
using System.Text;
using UnityEditor;
using UnityEngine;
using SebraeLab.Missions;

namespace SebraeLab.Editor
{
    /// <summary>
    /// Regenera Alvo.png com os 5 anéis nos raios de ArcheryScore.Raios, para a pintura e a pontuação saírem
    /// da mesma tabela. A borda da imagem vale o raio do último anel (0,70 m). Relatório em Logs/alvo-texture.txt.
    /// </summary>
    public static class AlvoTexture
    {
        const string Caminho = "Assets/SebraeLab/Art/Textures/Acabamentos/Alvo.png";
        const string Relatorio = "Logs/alvo-texture.txt";
        const int Lado = 512;

        static readonly Color Fundo = new Color32(0x2A, 0x4F, 0x54, 255);
        // Do centro para fora, na ordem dos anéis de ArcheryScore (10, 8, 6, 4 e 2 pontos).
        static readonly Color[] Aneis =
        {
            new Color32(0xEF, 0xB8, 0x5E, 255), new Color32(0xB4, 0x50, 0x35, 255), new Color32(0xEB, 0xDF, 0xBE, 255),
            new Color32(0x27, 0x6C, 0x74, 255), new Color32(0xEB, 0xDF, 0xBE, 255),
        };

        [MenuItem("Sebrae Lab/Arte/Gerar textura do alvo")]
        static void Menu() => Debug.Log("AlvoTexture: " + (Gerar() ? "OK" : "FALHA") + ". Detalhes em " + Relatorio);

        /// <summary>Para -executeMethod: fecha o Unity com 0 (ok) ou 1 (falha).</summary>
        public static void GerarPelaLinhaDeComando() => EditorApplication.Exit(Gerar() ? 0 : 1);

        public static bool Gerar()
        {
            var texto = new StringBuilder();
            int falhas = 0;
            void Check(bool ok, string mensagem)
            {
                if (!ok) falhas++;
                texto.AppendLine((ok ? "OK    " : "FALHA ") + mensagem);
            }

            try
            {
                float raioMaximo = ArcheryScore.Raios[ArcheryScore.Raios.Length - 1];
                var pixels = new Color[Lado * Lado];
                for (int y = 0; y < Lado; y++)
                    for (int x = 0; x < Lado; x++)
                    {
                        var soma = Color.clear;
                        foreach (float sy in new[] { .25f, .75f })
                            foreach (float sx in new[] { .25f, .75f })
                            {
                                float dx = (x + sx) / Lado * 2f - 1f, dy = (y + sy) / Lado * 2f - 1f;
                                soma += CorEm(Mathf.Sqrt(dx * dx + dy * dy) * raioMaximo);
                            }
                        pixels[y * Lado + x] = soma / 4f;
                    }
                var textura = new Texture2D(Lado, Lado, TextureFormat.RGBA32, false);
                textura.SetPixels(pixels);
                var bytes = textura.EncodeToPNG();
                UnityEngine.Object.DestroyImmediate(textura);
                File.WriteAllBytes(Caminho, bytes);
                AssetDatabase.ImportAsset(Caminho, ImportAssetOptions.ForceUpdate);
                texto.AppendLine($"PNG gravado em {Caminho} ({bytes.Length} bytes)");

                // Confere lendo o arquivo de volta: a cor no meio de cada faixa, de cada lado de uma borda e no canto.
                var lida = new Texture2D(2, 2, TextureFormat.RGBA32, false);
                Check(lida.LoadImage(File.ReadAllBytes(Caminho)) && lida.width == Lado && lida.height == Lado, $"PNG lido de volta com {lida.width} x {lida.height}");
                Color Amostra(float raio) => lida.GetPixel(Lado / 2 + Mathf.RoundToInt(raio / raioMaximo * (Lado / 2f)), Lado / 2);
                bool Igual(Color a, Color b) => Mathf.Abs(a.r - b.r) < .02f && Mathf.Abs(a.g - b.g) < .02f && Mathf.Abs(a.b - b.b) < .02f;
                float interno = 0f;
                for (int i = 0; i < ArcheryScore.Raios.Length; i++)
                {
                    float meio = (interno + ArcheryScore.Raios[i]) / 2f;
                    Check(Igual(Amostra(meio), Aneis[i]), $"anel de {ArcheryScore.PontosDosAneis[i]} pontos tem a cor esperada em {meio:0.###} m");
                    interno = ArcheryScore.Raios[i];
                }
                Check(!Igual(Amostra(.13f), Amostra(.15f)), "a cor muda na borda de 0,14 m");
                Check(Igual(lida.GetPixel(2, 2), Fundo), "o canto da imagem tem a cor do fundo");
                UnityEngine.Object.DestroyImmediate(lida);
            }
            catch (Exception e)
            {
                falhas++;
                texto.AppendLine("FALHA exceção: " + e);
            }

            texto.Insert(0, "Textura do alvo • " + DateTime.Now + "\n");
            texto.AppendLine(falhas == 0 ? "RESULTADO: OK" : $"RESULTADO: FALHA ({falhas})");
            Directory.CreateDirectory("Logs");
            File.WriteAllText(Relatorio, texto.ToString());
            Debug.Log(texto.ToString());
            return falhas == 0;
        }

        static Color CorEm(float raio)
        {
            for (int i = 0; i < ArcheryScore.Raios.Length; i++) if (raio <= ArcheryScore.Raios[i]) return Aneis[i];
            return Fundo;
        }
    }
}
```

```csharp
using System;
using System.Collections;
using System.IO;
using System.Linq;
using System.Text;
using UnityEditor;
using UnityEngine;
using UnityEngine.XR.Interaction.Toolkit;
using UnityEngine.XR.Interaction.Toolkit.Interactables;
using UnityEngine.XR.Interaction.Toolkit.Interactors;
using SebraeLab.Missions;
using Object = UnityEngine.Object;

namespace SebraeLab.Editor
{
    /// <summary>
    /// Verificação em Play do arco: montagem, disparo do PC, gatilho do Quest, tiro no chão e recarga
    /// (e, com a corda, puxada, soltura e cancelamento). Mãos de teste são XRDirectInteractor inativos que só
    /// servem de "quem pegou ou soltou". Pedido: Logs/archery-check.request (conteúdo "sair" fecha o editor ao
    /// terminar) ou menu. Resultado: Logs/archery-check.txt.
    /// </summary>
    [InitializeOnLoad]
    public static class ArcheryCheck
    {
        const string Request = "Logs/archery-check.request";
        const string Report = "Logs/archery-check.txt";
        const string ExitKey = "ArcheryCheck.Exit";
        const string PrefixoDasMaos = "Mão de teste (arco)";
        const float LimiteDoVoo = 8f;
        static readonly Vector3 PosicaoDoArco = new Vector3(51f, 1.5f, -2f);

        sealed class Relatorio
        {
            public readonly StringBuilder Texto = new StringBuilder();
            public int Falhas;
            public void Secao(string nome) => Texto.AppendLine().AppendLine("== " + nome);
            public void Check(bool ok, string m) { if (!ok) Falhas++; Texto.AppendLine((ok ? "OK    " : "FALHA ") + m); }
            public void Info(string m) => Texto.AppendLine("      " + m);
        }

        static ArcheryCheck() { EditorApplication.update += Poll; }

        [MenuItem("Sebrae Lab/Verificação/Arco")]
        static void Menu() { Directory.CreateDirectory("Logs"); File.WriteAllText(Request, ""); }

        static void Poll()
        {
            if (!File.Exists(Request) || EditorApplication.isCompiling || EditorApplication.isUpdating) return;
            if (!EditorApplication.isPlaying)
            {
                if (!EditorApplication.isPlayingOrWillChangePlaymode) EditorApplication.EnterPlaymode();
                return;
            }
            var mission = Object.FindFirstObjectByType<ManagerMission>();
            var alcance = Object.FindFirstObjectByType<ArcheryRange>();
            if (!mission || !alcance) return;
            SessionState.SetBool(ExitKey, File.ReadAllText(Request).Trim() == "sair");
            File.Delete(Request);
            mission.StartCoroutine(Run(alcance));
        }

        static IEnumerator Run(ArcheryRange alcance)
        {
            var r = new Relatorio();
            bool concluiu = false;
            r.Texto.AppendLine("Arco • " + DateTime.Now);
            var corpo = alcance.Arco.GetComponent<Rigidbody>();
            var posicaoInicial = corpo.position;
            var rotacaoInicial = corpo.rotation;
            try
            {
                Montagem(alcance, r);
                // Passos executados no mesmo iterador: uma exceção chega ao finally e o relatório sai mesmo assim.
                foreach (var passo in new Func<ArcheryRange, Relatorio, IEnumerator>[] { DisparoDoPc, Recarga, GatilhoDoQuest, TiroNoChao })
                {
                    var e = passo(alcance, r);
                    while (e.MoveNext()) yield return e.Current;
                }
                concluiu = true;
            }
            finally
            {
                if (!concluiu) r.Check(false, "execução interrompida por exceção; ver o Console");
                corpo.isKinematic = false;
                corpo.transform.SetPositionAndRotation(posicaoInicial, rotacaoInicial);
                corpo.position = posicaoInicial;
                corpo.rotation = rotacaoInicial;
                foreach (var mao in Object.FindObjectsByType<XRDirectInteractor>(FindObjectsInactive.Include, FindObjectsSortMode.None))
                    if (mao.name.StartsWith(PrefixoDasMaos)) Object.Destroy(mao.gameObject);
                r.Texto.AppendLine().AppendLine(r.Falhas == 0 ? "RESULTADO: OK" : $"RESULTADO: FALHA ({r.Falhas})");
                Directory.CreateDirectory("Logs");
                File.WriteAllText(Report, r.Texto.ToString());
                Debug.Log("ArcheryCheck\n" + r.Texto);
                if (SessionState.GetBool(ExitKey, false)) EditorApplication.Exit(r.Falhas == 0 ? 0 : 1);
            }
        }

        static void Montagem(ArcheryRange a, Relatorio r)
        {
            r.Secao("Montagem");
            r.Check(a.Arco && a.Alvo && a.FlechaCarregada, "arco, alvo e flecha carregada ligados ao ArcheryRange");
            Physics.SyncTransforms();
            var colisores = a.Alvo.GetComponents<Collider>();
            var caixa = colisores.OfType<BoxCollider>().FirstOrDefault(c => c.enabled);
            r.Check(caixa != null && !colisores.OfType<SphereCollider>().Any(c => c.enabled), "alvo com colisor de caixa e sem esfera ligada");
            if (caixa != null)
            {
                var t = caixa.bounds.size;
                r.Check(Mathf.Abs(t.x - 1.4f) < .01f && Mathf.Abs(t.y - 1.4f) < .01f && Mathf.Abs(t.z - .15f) < .01f, $"caixa do alvo com {t.x:0.###} x {t.y:0.###} x {t.z:0.###} m");
            }
            var face = a.Alvo.Find("Face do alvo");
            r.Check(face && Mathf.Abs(face.lossyScale.x - 1.4f) < .01f, $"face do alvo com {(face ? face.lossyScale.x : 0f):0.###} m de diâmetro");
            r.Check(a.Texto && a.Texto.text.StartsWith("ARCO ARCADE • 0 pontos"), $"placar inicial: \"{(a.Texto ? a.Texto.text.Replace("\n", " | ") : "")}\"");
            r.Check(a.FlechaCarregada && a.FlechaCarregada.transform.parent == a.Visual, "flecha de luz carregada sobre o arco recurvo");
        }

        static IEnumerator DisparoDoPc(ArcheryRange a, Relatorio r)
        {
            r.Secao("Disparo do PC");
            var centro = a.Alvo.position;
            Mirar(a, PosicaoDoArco, centro);
            yield return null;
            int antes = a.Placar.Total, tiros = a.Disparos;
            a.Arco.activated.Invoke(null); // o DesktopPlayer chama assim, sem argumentos
            yield return AteAcabar(a);
            var f = a.UltimaFlecha;
            r.Check(a.Disparos == tiros + 1 && f != null, $"um disparo ({a.Disparos - tiros})");
            if (f == null) yield break;
            r.Check(Mathf.Abs(f.VelocidadeInicial - ArcheryDraw.VelocidadeMaxima) < .05f, $"velocidade inicial {f.VelocidadeInicial:0.##} m/s");
            r.Check(f.AcertouAlvo, "a flecha parou no alvo");
            r.Check(Mathf.Abs(f.Ponta.z - (centro.z - .075f)) < .03f, $"parou na face do alvo a 40 m/s (z {f.Ponta.z:0.###}, face em {centro.z - .075f:0.###})");
            r.Check(a.Placar.UltimoPontos >= 8 && a.Placar.Total == antes + a.Placar.UltimoPontos,
                $"mirando o centro pontua 8 ou 10 (ganhou {a.Placar.UltimoPontos}, {a.Placar.UltimoRotulo}, a {f.DistanciaDoCentro:0.###} m do centro)");
            r.Check(a.Texto.text.StartsWith($"ARCO ARCADE • {a.Placar.Total} pontos"), $"placar mostra o total: \"{a.Texto.text.Replace("\n", " | ")}\"");
        }

        static IEnumerator Recarga(ArcheryRange a, Relatorio r)
        {
            r.Secao("Recarga");
            yield return AguardarFlecha(a);
            r.Check(a.FlechaCarregada && a.FlechaCarregada.transform.parent == a.Visual, "flecha recarregada sobre a corda");
        }

        static IEnumerator GatilhoDoQuest(ArcheryRange a, Relatorio r)
        {
            r.Secao("Gatilho do Quest");
            var mao = Mao("gatilho");
            int tiros = a.Disparos;
            a.Arco.activated.Invoke(new ActivateEventArgs { interactorObject = mao, interactableObject = a.Arco });
            yield return null;
            r.Check(a.Disparos == tiros && a.FlechaCarregada, "o gatilho do controle não dispara: a flecha só sai pela corda");
        }

        static IEnumerator TiroNoChao(ArcheryRange a, Relatorio r)
        {
            r.Secao("Tiro no chão");
            Mirar(a, PosicaoDoArco, new Vector3(PosicaoDoArco.x, 0f, 0f));
            yield return null;
            int antes = a.Placar.Total, tiros = a.Disparos;
            a.Arco.activated.Invoke(null);
            yield return AteAcabar(a);
            var f = a.UltimaFlecha;
            r.Check(a.Disparos == tiros + 1 && f != null && !f.Voando && !f.AcertouAlvo, "a flecha parou fora do alvo");
            r.Check(a.Placar.Total == antes && a.Placar.UltimoRotulo == "Errou", $"conta Errou sem mexer no total ({a.Placar.UltimoRotulo}, total {a.Placar.Total}, antes {antes})");
            r.Check(a.Texto.text.Contains("Último tiro: Errou"), $"placar mostra o erro: \"{a.Texto.text.Replace("\n", " | ")}\"");
            yield return AguardarFlecha(a);
        }

        // Ajudantes ---------------------------------------------------------------------------------------

        /// <summary>Põe o arco parado em <paramref name="de"/>, com o eixo +Z apontando para <paramref name="para"/>.</summary>
        static void Mirar(ArcheryRange a, Vector3 de, Vector3 para)
        {
            var corpo = a.Arco.GetComponent<Rigidbody>();
            corpo.isKinematic = true;
            var rotacao = Quaternion.LookRotation(para - de);
            corpo.transform.SetPositionAndRotation(de, rotacao);
            corpo.position = de;
            corpo.rotation = rotacao;
            Physics.SyncTransforms();
        }

        static IEnumerator AteAcabar(ArcheryRange a)
        {
            for (float t = 0; t < LimiteDoVoo && a.UltimaFlecha != null && a.UltimaFlecha.Voando; t += Time.deltaTime) yield return null;
            yield return null;
        }

        static IEnumerator AguardarFlecha(ArcheryRange a)
        {
            for (float t = 0; t < 2f && !a.FlechaCarregada; t += Time.deltaTime) yield return null;
            yield return null;
        }

        /// <summary>Mão de teste: objeto inativo, o interactor não acorda nem tenta selecionar nada.</summary>
        static XRDirectInteractor Mao(string nome)
        {
            var objeto = new GameObject(PrefixoDasMaos + " • " + nome);
            objeto.SetActive(false);
            return objeto.AddComponent<XRDirectInteractor>();
        }

        static SelectEnterEventArgs Entrar(IXRSelectInteractor mao, IXRSelectInteractable alvo) =>
            new SelectEnterEventArgs { interactorObject = mao, interactableObject = alvo };

        static SelectExitEventArgs Sair(IXRSelectInteractor mao, IXRSelectInteractable alvo) =>
            new SelectExitEventArgs { interactorObject = mao, interactableObject = alvo };
    }
}
```

```csharp
using System;
using UnityEngine;

namespace SebraeLab.Missions
{
    /// <summary>
    /// Flecha em voo. Integra a gravidade real e varre o trecho de cada passo com um raio saindo da ponta:
    /// a 40 m/s a flecha anda quase 1 m por quadro e o alvo tem 15 cm de espessura, então a colisão por evento
    /// poderia ser perdida. O voo roda em Update, em passos de no máximo 1/60 s, para ficar fluido no headset.
    /// </summary>
    public class ArcheryArrow : MonoBehaviour
    {
        /// <summary>Posição da ponta no espaço local da flecha de luz (a estrela termina em z 0,53).</summary>
        public static readonly Vector3 PontaLocal = new Vector3(0f, 0f, .53f);
        const float PassoMaximo = 1f / 60f;

        public bool Voando { get; private set; }
        public bool AcertouAlvo { get; private set; }
        public float VelocidadeInicial { get; private set; }
        public Vector3 Velocidade { get; private set; }
        public Vector3 Ponta { get; private set; }
        public float DistanciaDoCentro { get; private set; } = -1f;
        public float TempoNoAr { get; private set; }

        Transform alvo, arco;
        float tempoMaximo, cravadaPor;
        Action<float> acertou;
        Action errou;

        public void Lancar(Vector3 velocidade, Transform alvoDoTiro, Transform arcoDoTiro, float tempoMaximoNoAr, float cravadaPorSegundos,
            Action<float> aoAcertarAlvo, Action aoErrar)
        {
            alvo = alvoDoTiro;
            arco = arcoDoTiro;
            tempoMaximo = tempoMaximoNoAr;
            cravadaPor = cravadaPorSegundos;
            acertou = aoAcertarAlvo;
            errou = aoErrar;
            Velocidade = velocidade;
            VelocidadeInicial = velocidade.magnitude;
            Ponta = transform.TransformPoint(PontaLocal);
            Voando = true;
        }

        void Update()
        {
            if (!Voando) return;
            float restante = Mathf.Min(Time.deltaTime, .1f);
            while (Voando && restante > 0f)
            {
                float passo = Mathf.Min(restante, PassoMaximo);
                restante -= passo;
                Avancar(passo);
            }
        }

        void Avancar(float passo)
        {
            TempoNoAr += passo;
            Velocidade += Physics.gravity * passo;
            var deslocamento = Velocidade * passo;
            float comprimento = deslocamento.magnitude;
            if (comprimento > 1e-5f)
            {
                var acertos = Physics.RaycastAll(Ponta, deslocamento / comprimento, comprimento, ~0, QueryTriggerInteraction.Ignore);
                Array.Sort(acertos, (a, b) => a.distance.CompareTo(b.distance));
                foreach (var acerto in acertos)
                {
                    if (Ignorado(acerto.collider)) continue;
                    Parar(acerto.point, acerto.collider);
                    return;
                }
            }
            Ponta += deslocamento;
            var rotacao = Quaternion.LookRotation(Velocidade, Vector3.up);
            transform.SetPositionAndRotation(Ponta - rotacao * PontaLocal, rotacao);
            if (TempoNoAr > tempoMaximo || Ponta.y < -5f) Parar(Ponta, null);
        }

        bool Ignorado(Collider colisor) =>
            colisor is CharacterController || colisor.transform.IsChildOf(transform) || (arco && colisor.transform.IsChildOf(arco));

        void Parar(Vector3 ponto, Collider colisor)
        {
            Voando = false;
            Ponta = ponto;
            transform.position = ponto - transform.rotation * PontaLocal;
            bool noAlvo = colisor && alvo && (colisor.transform == alvo || colisor.transform.IsChildOf(alvo));
            if (noAlvo)
            {
                AcertouAlvo = true;
                DistanciaDoCentro = Vector3.ProjectOnPlane(ponto - alvo.position, alvo.forward).magnitude;
                acertou?.Invoke(DistanciaDoCentro);
            }
            else errou?.Invoke();
            Destroy(gameObject, cravadaPor);
        }
    }
}
```

```csharp
using System;
using UnityEngine;
using UnityEngine.XR.Interaction.Toolkit;
using UnityEngine.XR.Interaction.Toolkit.Interactables;

namespace SebraeLab.Missions
{
    /// <summary>
    /// Regras do arco da sala de descompressão: colisor do alvo, placar (<see cref="ArcheryScore"/>), disparo,
    /// disparo direto do modo PC e recarga. O InternshipDay monta as peças e chama <see cref="Configurar"/>.
    /// </summary>
    public class ArcheryRange : MonoBehaviour
    {
        [Header("Tempos (s)")]
        public float recarga = .35f;
        public float tempoCravada = 4f;
        public float tempoMaximoNoAr = 5f;

        [Header("Modo PC")]
        public float velocidadeDoPc = ArcheryDraw.VelocidadeMaxima;
        public float avancoDaFlechaNoPc = .38f;

        public ArcheryScore Placar { get; } = new ArcheryScore();
        public TextMesh Texto { get; private set; }
        public XRGrabInteractable Arco { get; private set; }
        public Transform Visual { get; private set; }
        public Transform Alvo { get; private set; }
        public Transform Area { get; private set; }
        public GameObject FlechaCarregada { get; private set; }
        public ArcheryArrow UltimaFlecha { get; private set; }
        public int Disparos { get; private set; }
        /// <summary>Verdadeiro enquanto alguma mão (ou o modo PC) segura o arco. Passar de uma mão para a outra não solta.</summary>
        public bool ArcoNaMao { get; private set; }

        /// <summary>Uma flecha nova foi montada sobre a corda.</summary>
        public event Action FlechaMontada;
        /// <summary>O arco deixou de ser segurado.</summary>
        public event Action ArcoSolto;

        Func<GameObject> montarFlecha;
        float recargaEm;

        public void Configurar(XRGrabInteractable arco, Transform visual, Func<GameObject> novaFlecha, GameObject alvo, TextMesh placar, Transform area)
        {
            Arco = arco;
            Visual = visual;
            Alvo = alvo.transform;
            Area = area;
            Texto = placar;
            montarFlecha = novaFlecha;

            // A esfera achatada vira uma bola de 0,7 m de raio com colisor de esfera; o alvo precisa de um disco fino.
            foreach (var colisor in alvo.GetComponents<Collider>()) colisor.enabled = false;
            alvo.AddComponent<BoxCollider>();
            Physics.SyncTransforms();

            FlechaCarregada = montarFlecha();
            arco.activated.AddListener(AoAtivar);
            arco.selectEntered.AddListener(AoPegarArco);
            arco.selectExited.AddListener(AoSoltarArco);
            AtualizarPlacar();
        }

        /// <summary>Solta a flecha carregada na direção +Z do arco, na pose em que ela está.</summary>
        public void Disparar(float velocidade)
        {
            if (!FlechaCarregada) return;
            var flecha = FlechaCarregada;
            FlechaCarregada = null;
            recargaEm = Time.time + recarga;
            Disparos++;
            flecha.transform.SetParent(Area, true);
            var voo = flecha.AddComponent<ArcheryArrow>();
            UltimaFlecha = voo;
            voo.Lancar(flecha.transform.forward * velocidade, Alvo, Arco.transform, tempoMaximoNoAr, tempoCravada, AoAcertar, AoErrar);
        }

        void Update()
        {
            if (FlechaCarregada || Time.time < recargaEm) return;
            FlechaCarregada = montarFlecha();
            FlechaMontada?.Invoke();
        }

        void AoAtivar(ActivateEventArgs args)
        {
            // Sem argumentos: modo PC (o DesktopPlayer chama activated.Invoke(null)). O gatilho do Quest traz
            // argumentos e é ignorado, para a flecha só sair pela corda.
            if (args != null || !FlechaCarregada) return;
            var arco = Arco.transform;
            FlechaCarregada.transform.SetPositionAndRotation(arco.position + arco.forward * avancoDaFlechaNoPc, arco.rotation);
            Disparar(velocidadeDoPc);
        }

        void AoPegarArco(SelectEnterEventArgs args) => ArcoNaMao = true;

        void AoSoltarArco(SelectExitEventArgs args)
        {
            // Passar o arco de uma mão para a outra não é soltar. args pode ser nulo (modo PC).
            if (Arco.isSelected) return;
            ArcoNaMao = false;
            ArcoSolto?.Invoke();
        }

        void AoAcertar(float distancia)
        {
            Placar.Registrar(distancia);
            AtualizarPlacar();
        }

        void AoErrar()
        {
            Placar.Errou();
            AtualizarPlacar();
        }

        void AtualizarPlacar()
        {
            if (Texto && Texto.text != Placar.Texto) Texto.text = Placar.Texto;
        }
    }
}
```

```csharp
using UnityEngine;
using UnityEngine.XR.Interaction.Toolkit;
using UnityEngine.XR.Interaction.Toolkit.Inputs.Haptics;
using UnityEngine.XR.Interaction.Toolkit.Interactables;
using UnityEngine.XR.Interaction.Toolkit.Interactors;

namespace SebraeLab.Missions
{
    /// <summary>
    /// Corda do arco. Um pegador (XRGrabInteractable pequeno e cinemático) fica no ponto de repouso da corda e só
    /// pode ser agarrado com o arco na mão. A mão livre puxa o pegador para trás, a flecha e a corda visual
    /// acompanham, e soltar dispara se a puxada passar do mínimo (<see cref="ArcheryDraw"/>).
    /// </summary>
    public class ArcheryString : MonoBehaviour
    {
        [Header("Geometria (espaço local do Arco recurvo)")]
        public Vector3 repouso = new Vector3(0f, 0f, -.12f);
        public float alturaDasPontas = .77f;
        public float espessuraDaCorda = .005f;
        public float raioDoPegador = .07f;

        [Header("Vibração")]
        public float forcaDaVibracao = .5f;
        public float duracaoDaVibracao = .1f;

        public XRGrabInteractable Pegador { get; private set; }
        public Transform SegmentoSuperior { get; private set; }
        public Transform SegmentoInferior { get; private set; }
        public float Puxada { get; private set; }
        public bool PegadorAgarrado { get; private set; }
        /// <summary>Ponto de repouso da corda no mundo.</summary>
        public Vector3 PosicaoDeRepouso => visual.TransformPoint(repouso);

        ArcheryRange alcance;
        Transform visual;
        Rigidbody corpo;
        Collider colisor;
        Vector3 flechaEmRepouso;

        public void Configurar(ArcheryRange dono)
        {
            alcance = dono;
            visual = dono.Visual;
            flechaEmRepouso = dono.FlechaCarregada.transform.localPosition;

            // A corda existente vira o segmento de cima; uma cópia dela, o de baixo.
            SegmentoSuperior = visual.Find("Corda do arco");
            SegmentoInferior = Instantiate(SegmentoSuperior.gameObject, visual).transform;
            SegmentoInferior.name = SegmentoSuperior.name;

            var objeto = new GameObject("Pegador da corda");
            objeto.transform.SetParent(dono.Area, false);
            corpo = objeto.AddComponent<Rigidbody>();
            corpo.isKinematic = true;
            corpo.useGravity = false;
            var esfera = objeto.AddComponent<SphereCollider>();
            esfera.radius = raioDoPegador;
            esfera.isTrigger = true; // a flecha varre só colisores sólidos
            esfera.enabled = false;  // ligado só enquanto o arco está na mão (ver Update)
            colisor = esfera;
            Pegador = objeto.AddComponent<XRGrabInteractable>();
            Pegador.movementType = XRBaseInteractable.MovementType.Instantaneous;
            Pegador.trackRotation = false;
            Pegador.throwOnDetach = false;
            Pegador.selectEntered.AddListener(AoAgarrar);
            Pegador.selectExited.AddListener(AoSoltar);
            dono.ArcoSolto += AoArcoSolto;
            Atualizar();
        }

        // O XRI move o pegador na própria atualização (e de novo antes de renderizar); a restrição roda depois dele.
        void OnEnable() => Application.onBeforeRender += Atualizar;
        void OnDisable() => Application.onBeforeRender -= Atualizar;
        void LateUpdate() => Atualizar();

        void Update()
        {
            if (colisor) colisor.enabled = alcance.ArcoNaMao;
        }

        float Medir() => ArcheryDraw.Limitar(Vector3.Dot(Pegador.transform.position - PosicaoDeRepouso, -visual.forward));

        void Atualizar()
        {
            if (!Pegador || !visual) return;
            Puxada = PegadorAgarrado ? Medir() : 0f;
            var ponto = PosicaoDeRepouso - visual.forward * Puxada;
            Pegador.transform.SetPositionAndRotation(ponto, visual.rotation); // descarta o desvio lateral
            corpo.position = ponto;
            var noque = repouso + Vector3.back * Puxada;
            Segmento(SegmentoSuperior, new Vector3(0f, alturaDasPontas, repouso.z), noque);
            Segmento(SegmentoInferior, new Vector3(0f, -alturaDasPontas, repouso.z), noque);
            var flecha = alcance.FlechaCarregada;
            if (flecha) flecha.transform.localPosition = flechaEmRepouso + Vector3.back * Puxada;
        }

        void Segmento(Transform segmento, Vector3 de, Vector3 ate)
        {
            var direcao = ate - de;
            segmento.localPosition = (de + ate) / 2f;
            segmento.localRotation = Quaternion.FromToRotation(Vector3.up, direcao);
            segmento.localScale = new Vector3(espessuraDaCorda, direcao.magnitude, espessuraDaCorda);
        }

        void AoAgarrar(SelectEnterEventArgs args)
        {
            if (alcance.ArcoNaMao) PegadorAgarrado = true;
        }

        void AoSoltar(SelectExitEventArgs args)
        {
            if (!PegadorAgarrado) return; // cancelado porque o arco foi solto: nenhum tiro
            Atualizar(); // fixa a puxada e a pose da flecha no instante da soltura
            float puxada = Puxada;
            PegadorAgarrado = false;
            if (alcance.ArcoNaMao && alcance.FlechaCarregada && ArcheryDraw.Dispara(puxada))
            {
                Vibrar(args != null ? args.interactorObject : null);
                Vibrar(alcance.Arco.firstInteractorSelecting);
                alcance.Disparar(ArcheryDraw.Velocidade(puxada));
            }
            Atualizar(); // devolve o pegador e a corda ao repouso
        }

        void AoArcoSolto()
        {
            if (!PegadorAgarrado) return;
            PegadorAgarrado = false;
            if (Pegador.isSelected && Pegador.interactionManager != null)
                Pegador.interactionManager.CancelInteractableSelection((IXRSelectInteractable)Pegador);
            Atualizar();
        }

        void Vibrar(IXRInteractor interactor)
        {
            if (interactor == null) return;
            var vibrador = interactor.transform.GetComponentInParent<HapticImpulsePlayer>();
            if (vibrador) vibrador.SendHapticImpulse(forcaDaVibracao, duracaoDaVibracao);
        }
    }
}
```
